## 邻接表定义
采用索引+边集数组，以锻炼新学的东西，不必看
```cpp
struct Edge
{
	int u, v;
};
struct ALGraph
{
	//存放对应位置是s中的第几条边
	vector<vector<int>> e;
	//边集数组
	vector<Edge>s;

	int vnum;
};

void CreateALGraph(ALGraph& G)
{
	cin >> G.vnum;
	G.e.resize(G.vnum);
	for (int i = 0; i < G.vnum; i++)
	{	
		for (int j = 0; j < G.vnum; j++)
		{
			int data;
			cin >> data;
			if (data)
			{
				G.s.push_back(Edge{ i, j });
			}
		}
	}
	for (int i = 0; i < G.s.size(); i++)
	{
		G.e[G.s[i].u].push_back(i);
	}
}
```

## Topological Sorting主体
用队列的目的：一次性处理所有入度为0的顶点  
邻接表找入度的技巧：设立入度数组，下标表示法，表示对应顶点的入度数
```cpp
bool Topological_Sorting(ALGraph& G, vector<int>&re)
{
	//入度数组
	vector<int>indeg;
	indeg.assign(G.vnum, 0);
	for (auto i : G.s)
	{
		indeg[i.v]++;
	}

	//存放入度为0的顶点
	queue<int>q;

	for (int i=0;i<G.vnum;i++)
	{
		if (indeg[i] == 0)q.push(i);
	}
	while (!q.empty())
	{
		int top = q.front();
		q.pop();
		re.push_back(top);
		//去掉其所有出度
		for (const auto idx:G.e[top])
		{
			//入度为0的顶点的下一个出度顶点
			int nextout = G.s[idx].v;
			indeg[nextout]--;
			if (indeg[nextout] == 0)q.push(nextout);
		}
		
	}
	return re.size() == G.vnum;
}
```
