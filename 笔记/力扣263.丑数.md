丑数是质因数只有2,3,5的数，给一个整数判断是否是丑数  
很简单，可优化  
```cpp
bool isUgly(int n)
{
	if (n <= 0)return false;
	while (n % 5 == 0 && n != 1)n /= 5;
	while (n % 3 == 0 && n != 1)n /= 3;
	while (n % 2 == 0 && n != 1)n /= 2;
	return n == 1;
}
```
利用已有数组整合重复代码
```cpp
//去掉重复代码
bool isUgly(int n,int flag)
{
	if (n <= 0)return false;
	vector<int>factors = { 2,3,5 };
	for(auto factor:factors)
	{
		while(n%factor ==0)
		{
			n /= factor;
		} 
	}
	return n == 1;
}
```
位运算操作找2的幂
```cpp
//遇到除以2还可以位运算
bool isUgly(int n,char flag)
{
	if (n <= 0)return false;
	while (n % 5 == 0 && n != 1)n /= 5;
	while (n % 3 == 0 && n != 1)n /= 3;
	
	return (n & (n - 1)) == 0;
	return (n & -n) == n;
}
```
