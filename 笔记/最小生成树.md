
## prim算法

原因：大多数情况下邻接矩阵简单  
目前没有数据超级大需要省空间的地步  
邻接表配合堆优化才明显优于邻接矩阵，但我不会  
总结：以下代码用邻接表实现，指针的操作嵌套且容易混淆，很难

### 邻接表的定义
1. 循环内的指针解引用一定要有迭代器，不然就一直对第一个元素操作，详见代码
2. 链表的头插法应该先初始化新结点的权，再连线

```cpp
//边
struct ArcNode
{
	//这条边的终点
	int adjvex;
	//边权
	int weight;
	//同一个起点的其他边
	ArcNode* next;

//顶点
struct VexNode
{
	string data;
	ArcNode* first;

struct ALGraph
{
	VexNode* vertices;

};

int Local(const ALGraph& G,string data)
{
	for (int i = 0; i < G.vexnum ; i++)
	{
		if (G.vertices [i].data == data)
		{
			return i;
		}
	}
	return -1;
}
void CreateGraph(ALGraph& G)
{
	int v, a;
	cin >> v ;

	//初始化G
	G.vexnum = v;
	G.vertices = new VexNode[v];

	//初始化Vex
	for (int i = 0; i < G.vexnum; i++)
	{
		//千万不要写成G.vertices->data，这样只初始化第一个元素
		cin >> G.vertices[i].data;
		G.vertices[i].first = NULL;
	}

	cin >> a;
	G.arcnum = a;

	//初始化Arc
	for (int i = 0; i < G.arcnum; i++)
	{
		string data1, data2;
		int loc1, loc2;
		cin >> data1 >> data2;
		loc1 = Local(G, data1);
		loc2 = Local(G, data2);

		//头插
		ArcNode* newnode = new ArcNode;
		newnode->adjvex = loc2;
		cin >> newnode->weight;
		newnode->next = G.vertices[loc1].first;
		G.vertices[loc1].first = newnode;

		ArcNode* newnode2 = new ArcNode;
		newnode2->adjvex = loc1;

		newnode2->next=G.vertices[loc2].first;

	}
```

### prim主体
思路根据笔记大概就可以模拟出来  
问题：  
1. 第一次多写了一层循环，在更新距离数组的部分，错误的想要遍历所有结点进行更新->思维错误，新加入结点只影响其相关的结点的距离，所以只看新结点的所有相关结点即可  
2. 想明白一个变量要干什么，existing为了判断下标位置元素是否已有，是和素数表类似的用法
```cpp
void Prim(ALGraph& G,Result &re)
{
	Distance* distance = new Distance[G.vexnum];
	//初始化存放 已有点与各个点距离 的数组
	for (int i = 0; i < G.vexnum; i++)
	{
		distance[i].dis = MAXINT;
		distance[i].isjoined = 0;
	}
	
	//存放 下标对应的顶点 是否是已有的
	bool* existing = new bool[G.vexnum]();
	//加入第一个顶点，并更新相关数组内信息
	string start;
	cin >> start;
	int loc1 = Local(G, start);
	existing[loc1] =true ;
	distance[loc1].last = -1;
	distance[loc1].dis = 0;
	distance[loc1].isjoined  = 1;

	//指针，是最新加入的顶点的下标
	int j = loc1;
	//每次取一条最短的线，一定会加入一个新结点，循环次数可知
	for (int i = 1; i < G.vexnum; i++)
	{
		
		//根据新加入的结点带来的距离，更新已有的距离数组
		ArcNode *AdjNode=G.vertices[j].first;
		while (AdjNode)
		{
			//比较与邻接顶点的边的权与记录的权
			if (AdjNode->weight < distance[AdjNode->adjvex].dis)
			{
				distance[AdjNode->adjvex].dis = AdjNode->weight;
				distance[AdjNode->adjvex].last = j;
			}
			AdjNode = AdjNode->next;
		}
		
		//寻找未有顶点中距离最近的
		int min = MAXINT;
		int minone = 0;
		for (int k = 0; k < G.vexnum; k++)
		{
			if (!distance[k].isjoined && distance[k].dis < min)
			{
				min = distance[k].dis;
				minone = k;
			}
		}
		
		//更新迭代
		j = minone;
		distance[minone].isjoined = true;
		existing[minone] = true;

		//保存结果
		re.arr.push_back(Pair(distance[minone].last, minone, distance[minone].dis));
		re.sum += distance[minone].dis;
	}

}
```

## kruskal算法
1. vector容器的resize是扩容并填充0，容器为空也一样
2. vector容器的assign是舍弃原有数据，重新分配空间，并填充目标元素v.assign(大小,目标元素)
3. kruskal作用于整张图，不管是否连通，它对所有边处理，能加的就会加上，所以不联通会生成森林
4. 并查集的理解见下

### 并查集
并查集用于处理集合问题中的集合合并与比较是否重合  
是本问题判断是否成环的关键部分  
前提：线由两个顶点约束，最小生成树的线经过全部顶点，二者可以看做集合以进行比较  
核心：若要加入的线的两端点属于最小生成树点集合，那么加入它必成环
```cpp
//定义一个并查集，名字含义是父，他是一个集合的代表人
vector<int>fa;

//寻找集合的代表人
int find(int x)
{
	//只有代表人的父是自己
	//if (fa[x] == x)
	//	return x;
	//else
	//{
	// //路径压缩，每个元素的父都改为代表人，使查询只需要一步
	//	fa[x] = find(fa[x]);
	//	return	fa[x];
	//}

	return fa[x] == x ? x : (fa[x] = find(fa[x]));
}

//合并两个集合
void merge(int x, int y)
{
	fa[find(y)] = find(x);
}
```

### kruskal主函数
解决了重点，实现就很简单
```cpp
//为kruskal专门定义的类，存放所有边
struct Edge
{
	//u代表头，v代表尾
	int u, v, w;
};

bool lower(const Edge& e1,const Edge& e2)
{
	return e1.w < e2.w;
}
void kruskal(const AMGraph& G,Result&re)
{
	//初始化fa数组
	fa.assign(G.vexnum,0);
	for (int i=0;i<G.vexnum ;i++)
	{
		fa[i] = i;
	}

	vector<Edge>e;
	//resize会填充0，然后可以用下标，如果不用下标填充反而多余
	//e.resize(G.arcnum);
	for (int i = 0; i < G.vexnum -1; i++)
	{
		for (int j = i+1; j<G.vexnum ;j++)
		{
			int v = G.arcs[i][j];
			if(v!=MAXINT)
				e.push_back(Edge(i, j, v));
		}
	}

	//测试边是否能加入
	//for (auto i : e)
	//{
	//	cout << i.u << " " << i.v << endl;
	//}

	sort(e.begin(), e.end(),lower);
	for (auto& i : e)
	{
		if (find(i.u) != find(i.v))
		{
			merge(i.u, i.v);
			re.arr.push_back(Pair(i.u, i.v, i.w));
		}
	}

	//判断非连通：选择的边数小于顶点数-1
	if (re.arr.size() < G.vexnum - 1)
		cout << "非连通图，生成森林" << endl;
}
```
